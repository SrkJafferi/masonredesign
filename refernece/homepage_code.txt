function playAdhaan(region){ // <<< do not edit or remove this line!

	// Get today's prayer times ("h:mm")
	var fajr=mytime(region,1);
	var zuhr=mytime(region,3);
	var maghrib=mytime(region,5);

	// Get current time
	var d=new Date();
	var hrs= parseInt(d.getHours());
	var mins = addZero(d.getMinutes());

	// Check if it's midday or morning
	if ( hrs > 11 && hrs < 24 ) {   // ZUHR OR MAGHRIB
		if (hrs != 12) {
			hrs = hrs - 12;		// Convert away from military hrs
		}

		var current_time = String(hrs) + ":" + String(mins);

		if (current_time == zuhr || current_time == maghrib){
			if (document.getElementById('adhaan').paused == true && !OFF) {
				document.getElementById('adhaan').play();
				OFF = true; // Prevents adhaan from playing again immediately if paused.
			}
		}
	} else {
		// Fajr
		var current_time = String(hrs) + ":" + String(mins);
		if (current_time == fajr){
			// If adhaan is not already playing and autoplay is on.
			if (document.getElementById('adhaan').paused == true && !OFF) {
				document.getElementById('adhaan').play();
				OFF = true;
			}
		}
	}
}
